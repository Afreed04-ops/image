# Final robust detector: multi-scale, rotation-aware, NMS, annotated output
import fitz         # PyMuPDF
import cv2
import numpy as np
import os
from matplotlib import pyplot as plt

# ---------- USER CONFIG ----------
pdf_path = r"C:/Bahjoi_SIP.pdf"
ref_paths = [r"C:/l5.jpg"]   # your two reference images
output_dir = r"/mnt/data/pdf_matches"
os.makedirs(output_dir, exist_ok=True)

dpi = 300                       # render DPI (increase to 400-600 for tiny templates)
scales = np.linspace(0.6, 1.4, 9)   # try 0.4..2.0 if shapes vary a lot
angles = [0, 90, 180, 270]      # rotations to test (add intermediate angles if needed)
threshold = 0.60                # template-matching threshold (lower to be more permissive)
nms_iou_thresh = 0.35           # IoU threshold for non-maximum suppression
max_peaks_per_scale = 8         # limit peaks per scale to speed up processing
# ---------------------------------

# load pdf page(s)
doc = fitz.open(pdf_path)
num_pages = len(doc)
print(f"✅ PDF loaded: {pdf_path} ({num_pages} pages)")

# We'll process each page and accumulate counts
total_counts = [0] * len(ref_paths)
annotated_pages = []

for page_idx in range(num_pages):
    page = doc.load_page(page_idx)
    pix = page.get_pixmap(dpi=dpi)
    img_data = np.frombuffer(pix.tobytes("png"), np.uint8)
    page_bgr = cv2.imdecode(img_data, cv2.IMREAD_COLOR)
    page_gray = cv2.cvtColor(page_bgr, cv2.COLOR_BGR2GRAY)

    print(f"\n🔍 Scanning page {page_idx+1}/{num_pages} (render: {pix.width}x{pix.height} px)")

    # To draw boxes for preview
    annotated = page_bgr.copy()

    for ref_idx, ref_path in enumerate(ref_paths):
        ref_orig = cv2.imread(ref_path, cv2.IMREAD_GRAYSCALE)
        if ref_orig is None:
            raise FileNotFoundError(f"Reference image not found: {ref_path}")

        # optional binarization helps for simple black-line drawings:
        _, ref_bin = cv2.threshold(ref_orig, 200, 255, cv2.THRESH_BINARY)

        detections = []  # list of (x,y,w,h,score)

        for angle in angles:
            # rotate template
            if angle != 0:
                # rotate keeping same size canvas
                (h0, w0) = ref_bin.shape
                M = cv2.getRotationMatrix2D((w0 / 2, h0 / 2), angle, 1.0)
                rotated = cv2.warpAffine(ref_bin, M, (w0, h0), borderValue=255)
            else:
                rotated = ref_bin

            for scale in scales:
                tw = max(2, int(rotated.shape[1] * scale))
                th = max(2, int(rotated.shape[0] * scale))
                if tw >= page_gray.shape[1] or th >= page_gray.shape[0]:
                    continue

                template = cv2.resize(rotated, (tw, th), interpolation=cv2.INTER_AREA if scale < 1 else cv2.INTER_LINEAR)

                # run template match
                res = cv2.matchTemplate(page_gray, template, cv2.TM_CCOEFF_NORMED)

                # get coordinates of peaks above threshold
                loc = np.where(res >= threshold)
                # gather (score, (x,y))
                peaks = []
                for (x, y) in zip(*loc[::-1]):
                    peaks.append((res[y, x], x, y))
                # sort peaks by score descending and keep top K to avoid flooding
                peaks = sorted(peaks, key=lambda p: -p[0])[:max_peaks_per_scale]

                for score, x, y in peaks:
                    detections.append((int(x), int(y), int(tw), int(th), float(score)))

        # Non-maximum suppression (greedy by score)
        detections = sorted(detections, key=lambda d: -d[4])
        kept = []
        for det in detections:
            x, y, w, h, sc = det
            box = np.array([x, y, x + w, y + h])
            suppress = False
            for k in kept:
                kx, ky, kw, kh, ksc = k
                kbox = np.array([kx, ky, kx + kw, ky + kh])
                # compute IoU
                ix1 = max(box[0], kbox[0]); iy1 = max(box[1], kbox[1])
                ix2 = min(box[2], kbox[2]); iy2 = min(box[3], kbox[3])
                iw = max(0, ix2 - ix1); ih = max(0, iy2 - iy1)
                inter = iw * ih
                area = (box[2] - box[0]) * (box[3] - box[1])
                area_k = (kbox[2] - kbox[0]) * (kbox[3] - kbox[1])
                union = area + area_k - inter
                iou = inter / union if union > 0 else 0
                if iou > nms_iou_thresh:
                    suppress = True
                    break
            if not suppress:
                kept.append(det)

        # record detections, draw boxes
        print(f"  • {os.path.basename(ref_path)} -> raw detections: {len(detections)}, distinct after NMS: {len(kept)}")
        total_counts[ref_idx] += len(kept)
        for (x, y, w, h, sc) in kept:
            cv2.rectangle(annotated, (x, y), (x + w, y + h), (0, 0, 255), 2)
            # label with score
            cv2.putText(annotated, f"{sc:.2f}", (x, y - 4), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0,0,255), 1)

    # save annotated preview for this page
    out_img = os.path.join(output_dir, f"page_{page_idx+1}_annotated.png")
    cv2.imwrite(out_img, annotated)
    annotated_pages.append(out_img)
    print(f"  → Annotated preview saved: {out_img}")

doc.close()

# Final summary
total = sum(total_counts)
print("\n🎯 Final counts:")
for i, c in enumerate(total_counts):
    print(f"  {os.path.basename(ref_paths[i])}: {c} matches")
print(f"  TOTAL (sum): {total}")
